Permission Check Decorator


# from functools import wraps
# import discord
# from utils.permissions import PermissionLevel, PermissionChecker

def require_permission(level: PermissionLevel):
    """
    Decorator for commands that checks if the user has the required permission level.
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(self, ctx, *args, **kwargs):
            if not await PermissionChecker.check_permission(ctx.author, level):
                embed = discord.Embed(
                    title="❌ Insufficient Permission",
                    description=f"You need {level.name} permission to use this command.",
                    color=discord.Color.red()
                )
                await ctx.send(embed=embed)
                return
            return await func(self, ctx, *args, **kwargs)
        return wrapper
    return decorator

Message-based Point Acquisition


# from discord.ext import commands
# from datetime import datetime, timedelta

@commands.Cog.listener()
async def on_message(self, message):
    """Message-based point accumulation"""
    if message.author.bot:
        return
        
    user_id = message.author.id
    current_time = datetime.now()
    
    # Cooldown check (accumulate only once per minute)
    if user_id in self.message_cooldown:
        if current_time - self.message_cooldown[user_id] < timedelta(minutes=1):
            return
    
    self.message_cooldown[user_id] = current_time
    
    # Add points
    await self.add_points(user_id, self.point_rates['message'], "Message sent")
    
    # Experience system (level-up check)
    await self.check_level_up(user_id)

Scheduled Message Deletion Feature


# from discord.ext import tasks
# from datetime import datetime
# from database.models import ScheduledDeletionModel

@tasks.loop(minutes=1)
async def deletion_checker(self):
    """Check scheduled deletion tasks"""
    current_time = datetime.now()
    
    try:
        # Query messages to delete from database
        deletions = await ScheduledDeletionModel.get_pending_deletions(self.bot.db, current_time)
        
        for deletion in deletions:
            await self.process_scheduled_deletion(deletion)
    except Exception as e:
        self.bot.logger.error(f"Error in deletion checker: {e}", exc_info=True)

Message Deletion Logging


# from discord.ext import commands
# import discord
# from datetime import datetime

@commands.Cog.listener()
async def on_message_delete(self, message):
    """Message deletion logging"""
    if message.author.bot:
        return
        
    # Send deletion information to log channel
    if self.bot.log_channel:
        embed = discord.Embed(
            title="🗑️ Message Deleted",
            description=message.content[:1000] if message.content else "*No content*",
            color=discord.Color.red(),
            timestamp=datetime.now()
        )
        embed.set_author(
            name=f"{message.author.display_name} ({message.author})",
            icon_url=message.author.avatar.url if message.author.avatar else None
        )
        embed.add_field(name="Channel", value=message.channel.mention, inline=True)
        embed.add_field(name="Message ID", value=message.id, inline=True)
        
        await self.bot.log_channel.send(embed=embed)

Error Handling Implementation


def error_handler(func):
    """
    Decorator that handles common errors for commands.
    """
    @wraps(func)
    async def wrapper(self, ctx, *args, **kwargs):
        try:
            return await func(self, ctx, *args, **kwargs)
        except commands.MissingRequiredArgument as e:
            # Missing argument
            embed = discord.Embed(
                title="❓ Missing Argument",
                description=f"Required argument missing: {e.param.name}",
                color=discord.Color.yellow()
            )
            embed.add_field(
                name="Usage", 
                value=f"`{ctx.prefix}{ctx.command.qualified_name} {ctx.command.signature}`",
                inline=False
            )
            await ctx.send(embed=embed)
        except Exception as e:
            # Log the error
            self.bot.logger.error(f"Command error in {ctx.command}: {e}", exc_info=True)
            # Notify user
            await ctx.send(f"An error occurred: {str(e)}")
    return wrapper
